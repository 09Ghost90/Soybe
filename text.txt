# ---------- Início: cabeçalho / instruções ----------
# Backup antes de alterar!
# Se o cPanel gerou directives php (MultiPHP INI), mantenha-as se precisar.
# ---------- Fim do cabeçalho ----------

# Desativa listagem de diretórios
Options -Indexes

# Protege arquivos sensíveis
<FilesMatch "^(?:wp-config\.php|readme\.html|license\.txt|xmlrpc\.php)$">
  Order allow,deny
  Deny from all
</FilesMatch>

# Protege o .htaccess e arquivos ocultos (exceto .well-known para SSL)
<FilesMatch "^\.">
  <If "%{REQUEST_URI} !~ m#^/\.well-known/#">
    Require all denied
  </If>
</FilesMatch>

# Bloqueia métodos HTTP perigosos
<IfModule mod_rewrite.c>
  RewriteEngine On
  # Permitir apenas GET, POST, HEAD
  RewriteCond %{REQUEST_METHOD} !^(GET|POST|HEAD)$ [NC]
  RewriteRule .* - [F,L]
</IfModule>

# Regras para bloquear strings maliciosas na query (injeção comum)
<IfModule mod_rewrite.c>
  RewriteCond %{QUERY_STRING} (base64_encode|base64_decode|eval\(|gzinflate|GLOBALS|_REQUEST|cmd=) [NC]
  RewriteRule .* - [F,L]
</IfModule>

# Evita execução de PHP em uploads via rewrite
<IfModule mod_rewrite.c>
  RewriteEngine On
  # Bloqueia requisições para arquivos .php dentro de wp-content/uploads
  RewriteRule ^wp-content/uploads/.*\.(php|phtml|phar)$ - [F,NC,L]
</IfModule>

# Protege /wp-includes/ (bloqueia acesso direto a arquivos PHP exceto admin-ajax.php)
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteRule ^wp-includes/.*\.php$ - [F,L,NC]
  RewriteRule ^wp-includes/js/tinymce/wp-tinymce.php - [L,NC]
  RewriteRule ^wp-includes/ms-files.php - [L,NC]
  RewriteRule ^wp-includes/[^/]+\.php$ - [F,L,NC]
</IfModule>

# Bloqueia hotlinking básico (opcional)
<IfModule mod_rewrite.c>
  RewriteCond %{HTTP_REFERER} !^$
  RewriteCond %{HTTP_REFERER} !^https?://(www\.)?%{HTTP_HOST} [NC]
  RewriteRule \.(jpg|jpeg|png|gif|svg|webp)$ - [F,NC,L]
</IfModule>

# Cabeçalhos de segurança (somente Apache 2.2+/mod_headers)
<IfModule mod_headers.c>
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-XSS-Protection "1; mode=block"
  # HSTS (enable se tiver HTTPS em todo site)
  # Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
</IfModule>

# Cache e expiração básica para assets (opcional)
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/jpg "access plus 30 days"
  ExpiresByType image/jpeg "access plus 30 days"
  ExpiresByType image/gif "access plus 30 days"
  ExpiresByType image/png "access plus 30 days"
  ExpiresByType text/css "access plus 7 days"
  ExpiresByType application/javascript "access plus 7 days"
  ExpiresByType text/javascript "access plus 7 days"
  ExpiresByType text/html "access plus 1 hour"
</IfModule>

# ================= WordPress padrão (permalinks) =================
# BEGIN WordPress
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /
RewriteRule ^index\.php$ - [L]

# Condições para arquivos/dirs existentes
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d

RewriteRule . /index.php [L]
</IfModule>
# END WordPress
